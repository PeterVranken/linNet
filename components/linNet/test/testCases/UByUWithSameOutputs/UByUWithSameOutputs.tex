\documentclass[tikz]{standalone} % Standalone: Only graphics export.
\usepackage[european,siunitx]{circuitikz}
\usetikzlibrary{arrows.meta}
\usepackage{listings}
\lstset{basicstyle=\footnotesize\ttfamily}
\begin{document}

  \newcommand{\eq}{=}
  \tikzset{ameter/.style={rmeterwa, t=A}}
  \begin{circuitikz}

    \tikzset{>=Triangle} % Define tip of arrows drawn with [->]
    \ctikzset{voltage=straight}

    \def\size{5}
    \def\deltaUio{0.8}
    \def\gapUin{0.3}
    \def\lR{2.5}

    \draw(0,\lR) coordinate(in1) node[above]{$n_1$}
      to[short,o-*] ++(\deltaUio,0) coordinate(LftUu)
      to[short,-*] ++(\lR,0) coordinate(RhtUu)
      to[R,l=$R_1$] ++(\lR,0)
      to[R,l=$R_2$,-*] ++(0,-\lR) coordinate(gnd) node[ground]{};
    \draw(0,0) coordinate(in1Gnd)
      to[short,o-] (gnd)
      to[short] ++(0.5*\lR,0)
      to[short,-o] ++(\deltaUio,0) coordinate(in2Gnd);
    \draw ($(in2Gnd)+(0,\lR)$) coordinate(in2)
      node[ocirc,label=above:$n_3$]{};
      
    \draw (LftUu)
      -- ++(0,0.5*\lR)
      to[V,v<=$U(U)\eq k_U U_2$] ++(\lR,0)
      -- ++(0,-0.5*\lR);

    % Voltage arrows for input voltages U1 and U2.
    \draw[->] ($(in1)+(0,-\gapUin)$) -- node[midway,left]{$U_1$} ($(in1Gnd)+(0,\gapUin)$);
    \draw[->] ($(in2)+(0,-\gapUin)$) -- node[midway,right]{$U_2$} ($(in2Gnd)+(0,\gapUin)$);

  \end{circuitikz}

\end{document}
