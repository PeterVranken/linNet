\documentclass[tikz]{standalone} % Standalone: Only graphics export.
\usepackage[european,siunitx]{circuitikz}
\usetikzlibrary{arrows.meta}
\usepackage{listings}
\lstset{basicstyle=\footnotesize\ttfamily}
\begin{document}

  \begin{circuitikz}

    \tikzset{>=Triangle} % Define tip of arrows drawn with [->]
    \ctikzset{voltage=straight}

    \def\gapUin{0.2}
    \def\distance{1.5}
    
    % Connectors for input voltage Uin.
    \node[ocirc, label=above:{$K_1$}] at (0,0){};
    \draw (0,-1) to[short,o-] ++(0,-0.4) node[ground, label=left:{$gnd$}]{};

    % Voltage arrow for input voltage U_in.
    \draw[->] (0,-\gapUin) -- node[midway,left]{$U_{in}$} ($(0,-1)+(0,\gapUin)$);

    \draw (\distance,0.5) node[above]{$K_+$} to[V, v=$kU_{in}$] ++(0,-2) node[left]{$K_-$} to[R, l=$R_{gnd}$, -*] ++(0,-2)
      coordinate(gnd2) to[short] ++(2,0) to [R, l=$R_{out}$] ++(0,4)
      -- (\distance,0.5);

    \draw (gnd2) -- ++(0,-0.4) node[ground, label=left:{$gnd_2$}]{};

  \end{circuitikz}

\end{document}
