\documentclass[tikz]{standalone} % Standalone: Only graphics export.
\usepackage[european,siunitx]{circuitikz}
\usetikzlibrary{arrows.meta}
\usepackage{listings}
\lstset{basicstyle=\footnotesize\ttfamily}
\begin{document}

  \begin{circuitikz}

    \tikzset{>=Triangle} % Define tip of arrows drawn with [->]
    \ctikzset{voltage=straight}

    \def\gapUin{0.2}
    \def\distance{2.1}
    \def\deltaH{1}
    
    % Connectors for input voltage Uin.
    \node[ocirc, label=above:{$K_1$}] at (0,0){};
    \draw (0,-1) coordinate(gnd) to[short,o-] ++(0,-0.4) node[ground, label=left:{$gnd$}]{};

    % Voltage arrow for input voltage U_in.
    \draw[->] (0,-\gapUin) -- node[midway,left]{$U_{in}$} ($(0,-1)+(0,\gapUin)$);

    \draw (\distance,\deltaH) node[above]{$K_+$} to[V, v=$kU_{in}$]
      ++(0,-2) node[left]{$K_-$} to[R, l=$R_{gnd}$, -*] ++(0,-2)
      coordinate(gnd2) to[short] ++(2,0) to [R, l=$R_{out}$] ++(0,4)
      to[short,-*] (\distance,\deltaH) coordinate(Kplus);

    \draw (gnd2) -- ++(0,-0.4) node[ground, label=left:{$gnd_2$}]{};

    % Voltage arrow for bad voltage U2. 
    %   Note, the arrows for U2 and U3 hit the end nodes only roughly and
    % strongly depending on the values of \distance and \deltaH -- these
    % must better not be modified any more.
    \draw[->] ($(Kplus)+(-\gapUin,-\gapUin)$) -- node[midway,above left]{$U_2$}
      ($(gnd)+(\gapUin,\gapUin)$);

    % Voltage arrow for bad voltage U3.
    \draw[->] ($(gnd)+(\gapUin,-\gapUin)$) -- node[midway,below left]{$U_3$}
      ($(gnd2)+(-\gapUin,\gapUin)$);


  \end{circuitikz}

\end{document}
