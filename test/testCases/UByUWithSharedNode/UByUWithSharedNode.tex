  \tikzset{ameter/.style={rmeterwa, t=A}}
  \begin{circuitikz}

    \ctikzset{voltage=straight}

    \def\lR{2.5}

    \draw(0,\lR) coordinate(in0) node[above]{$K_1$}
      to[R,l=$R_1$,o-] ++(\lR,0) node[above]{$K_2$}
      to[R,l=$R_2$,v=$U_{K_2 K_3}$,-*] ++(\lR,0) node[above]{$K_3$}
      to[R,l=$R_3$,-*] ++(0,-\lR) coordinate(gnd) node[ground]{}
      to[short] ++(0.75*\lR,0)
      to[V,v_<=$U(U)\eq kU_{K_2 K_3}$] ++(0,\lR)
      to[short] ++(-0.75*\lR,0);

      to[short,o-] (gnd);

    % Voltage arrow for input voltage U0.
    \draw[->] ($(in0)+(0,-\gapUin)$) -- node[midway,left]{$U_0$} ($(in0Gnd)+(0,\gapUin)$);
